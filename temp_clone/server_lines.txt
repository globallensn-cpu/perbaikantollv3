   243	  async function await getClientInfoByCode(accessCode?: string): Promise<{ name: string; accessCode: string; email?: string }> {
   244	    if (!accessCode) return { name: 'Klien Satset', accessCode: 'GUEST' };

   259	  async function await getClientIsolatedKeys(customApiKeyHeader?: string, clientAccessCode?: string): Promise<string[]> {
   260	    let userCustomKeys: string[] = [];
   261	    if (customApiKeyHeader && customApiKeyHeader.trim()) {
   262	      userCustomKeys = customApiKeyHeader
   263	        .split(/[
   264	,]+/)
   265	        .map((k) => k.trim())
   266	        .filter(Boolean);
   267	    }
   268	
   269	    let boundKeys: string[] = [];
   270	    if (clientAccessCode && clientAccessCode !== 'GUEST') {
   271	      const cleanCode = clientAccessCode.trim().toUpperCase();
   272	      try {
   273	        const keysArr = await dbGetApiKeys();
   274	        boundKeys = keysArr
   275	          .filter((k: any) => k.status === 'active' && k.accessCode && k.accessCode.toUpperCase() === cleanCode)
   276	          .map((k: any) => k.key);
   277	      } catch (e) { logger.warn('[API Keys] Gagal membaca api keys untuk access code', e); }
   278	    }
   279	
   280	    const clientKeys = Array.from(new Set([...userCustomKeys, ...boundKeys])).filter(Boolean);
   281	    if (clientKeys.length > 0) {
   282	      return clientKeys;
   283	    }
   284	
   285	    const systemKey = process.env.GEMINI_API_KEY ? process.env.GEMINI_API_KEY.trim() : '';
   286	    let globalKeys: string[] = [];
   287	    try {
   288	      const keysArr = await dbGetApiKeys();
   289	      globalKeys = keysArr
   290	        .filter((k: any) => k.status === 'active' && (!k.accessCode || k.accessCode === 'SYSTEM' || k.accessCode === 'GLOBAL'))
   291	        .map((k: any) => k.key);
   292	    } catch (e) { logger.warn('[API Keys] Gagal membaca global api keys', e); }
   293	
   294	    return Array.from(new Set([systemKey, ...globalKeys])).filter(Boolean);
   295	  }

   308	  // Helper function for calling Gemini API with key rotation, candidate model fallback & anti-limit exponential backoff + jitter
   309	  async function callGeminiWithFallback(
   310	    userSelectedModel: string,
   311	    promptPayload: any,
   312	    customApiKeyHeader?: string,
   313	    clientAccessCode?: string,
   314	    targetTier?: any
   315	  ): Promise<{ text: string; modelUsed: string }> {
   316	    // Fallback to legacy isolated keys logic if custom key logic is bypassed
   317	    const keyCandidates = await getClientIsolatedKeys(customApiKeyHeader, clientAccessCode);
   318	    
   319	    // We will still support old array of candidate models as fallback
   320	    let candidateModels = Array.from(new Set([
   321	      userSelectedModel,
   322	      'gemini-2.5-flash',
   323	      'gemini-2.0-flash',
   324	      'gemini-1.5-flash',
   325	      'gemini-3.6-flash',
   326	      'gemini-2.5-flash-lite',
   327	      'gemini-3.5-flash-lite',
   328	      'gemini-3.1-flash-lite',
   329	      'gemini-2-flash-lite',
   330	      'gemini-2.5-pro',
   331	      'gemini-3.1-pro-preview',
   332	      'gemini-3.1-pro',
   333	      'gemini-1.5-pro'
   334	    ])).filter((m): m is string => Boolean(m && m.trim().length > 0));
   335	
   336	    // Calculate routing plan if targetTier provided
   337	    try {
   338	       const plan = getModelRoutingPlan(userSelectedModel, customApiKeyHeader);
   339	       if (plan && plan.targetModels && plan.targetModels.length > 0) {
   340	          if (targetTier) {
   341	              const tierModels = targetTier === 'flagship' ? MODEL_TIERS.flagship : targetTier === 'tier3' ? MODEL_TIERS.tier3 : MODEL_TIERS.tier2;
   342	              candidateModels = Array.from(new Set([
   343	                  userSelectedModel,
   344	                  ...tierModels,
   345	                  ...plan.targetModels
   346	              ]));
   347	          } else {
   348	             candidateModels = plan.targetModels;
   349	          }
   350	       }
   351	    } catch (e) {
   352	       logger.warn('[Routing] Failed to calculate model routing plan, using fallback list');
   353	    }
   354	
   355	    if (keyCandidates.length === 0) {
   356	      throw new Error('API Key Gemini tidak dikonfigurasi. Silakan atur di Pengaturan Anti Limit.');
   357	    }
   358	
   359	    let lastError: any = null;
   360	
   361	    for (let kIdx = 0; kIdx < keyCandidates.length; kIdx++) {
   362	      const activeKey = keyCandidates[kIdx];
   363	      let aiInstance: GoogleGenAI;
   364	      try {
   365	        aiInstance = getGeminiClient(activeKey);
   366	      } catch (e) {
   367	        continue;
   368	      }
   369	
   370	      for (const targetModel of candidateModels) {
   371	        let attempts = 0;
   372	        const maxAttempts = 2;
   373	
   374	        while (attempts < maxAttempts) {
   375	          attempts++;
   376	          try {
   377	            logger.info(`[Gemini Request] Key #${kIdx + 1}/${keyCandidates.length}, Model: ${targetModel}, Attempt: ${attempts}...`);
   378	            
   379	            // Build request config with High Thinking mode support for gemini-3.1-pro-preview
   380	            const isThinkingModel = targetModel === 'gemini-3.1-pro-preview' || targetModel === 'gemini-3.1-pro';
   381	            const baseConfig = promptPayload.config || {};
   382	            
   383	            const requestConfig: any = {
   384	              ...baseConfig,
   385	            };
   386	
   387	            const rawInstruction = requestConfig.systemInstruction || "You are an elite AI assistant.";
   388	            requestConfig.systemInstruction = getInjectedSystemInstruction(rawInstruction);
   389	
   390	            if (isThinkingModel) {
   391	              requestConfig.thinkingConfig = {
   392	                thinkingLevel: 'HIGH',
   393	              };
   394	              // Crucial: Do NOT set maxOutputTokens when thinking mode is enabled
   395	              delete requestConfig.maxOutputTokens;
   396	            }
   397	
   398	            const response = await aiInstance.models.generateContent({
   399	              model: targetModel,
   400	              contents: promptPayload.contents,
   401	              config: requestConfig,
   402	            });
   403	
   404	            if (response && response.text) {
   405	              return { text: response.text, modelUsed: targetModel };
   406	            }
   407	          } catch (err: any) {
   408	            lastError = err;
   409	            const errMsg = String(err?.message || err || '');
   410	            const status = (err as any)?.status || (err as any)?.statusCode || 0;
   411	
   412	            logger.warn(`[Gemini Warning] Key #${kIdx + 1}, Model ${targetModel} attempt ${attempts} failed:`, errMsg);
   413	
   414	            const isDeadKey =
   415	              status === 403 ||
   416	              errMsg.includes('403') ||
   417	              errMsg.includes('API_KEY_INVALID') ||
   418	              errMsg.includes('API key not found') ||
   419	              errMsg.includes('PERMISSION_DENIED') ||
   420	              errMsg.includes('UNAUTHENTICATED');
   421	
   422	            const isRateLimitOrQuota =
   423	              status === 429 ||
   424	              errMsg.includes('429') ||
   425	              errMsg.includes('RESOURCE_EXHAUSTED') ||
   426	              errMsg.includes('Quota exceeded') ||
   427	              errMsg.includes('limit: 0');
   428	
   429	            if (isDeadKey || isRateLimitOrQuota) {
   430	                // Apply circuit breaker: cooldown for 5 minutes
   431	                try {
   432	                   const keys = getApiKeys();
   433	const foundKey = keys.find((k: any) => k.key === activeKey);
   434	                   if (foundKey) {
   435	                       foundKey.cooldownUntil = Date.now() + 5 * 60 * 1000;
   436	                       saveApiKeys(keys);
   437	logger.warn(`[Circuit Breaker] Key #${kIdx + 1} isolated until ${new Date(foundKey.cooldownUntil).toISOString()}`);
   438	                   }
   439	                } catch(cbErr) {
   440	                   // ignore if admin tools not accessible
   441	                }
   442	            }
   443	
   444	            if (isDeadKey) {
   445	              logger.warn(`[Auto-Prune] Key #${kIdx + 1} is invalid/dead. Rotating to next key...`);
   446	              attempts = maxAttempts;
   447	              break;
   448	            }
   449	
   450	            if (isRateLimitOrQuota) {
   451	              logger.info(`[Model Cascading] Model ${targetModel} rate limited on Key #${kIdx + 1}. Cascading to next model...`);
   452	              attempts = maxAttempts;
   453	              break;
   454	            }
   455	
   456	            if (errMsg.includes('503') || errMsg.includes('UNAVAILABLE') || errMsg.includes('demand')) {
   457	              const jitter = Math.floor(Math.random() * 2000);
   458	              const backoffDelay = 1000 + jitter;
   459	              logger.info(`[Anti-Limit Backoff] Waiting ${backoffDelay}ms before retry...`);
   460	              await new Promise((resolve) => setTimeout(resolve, backoffDelay));
   461	            } else {
   462	              attempts = maxAttempts;
   463	              break;
   464	            }
   465	          }
   466	        }
   467	
   468	        const lastErrStr = String(lastError?.message || lastError || '');
   469	        const isDeadKey =
   470	          lastErrStr.includes('403') ||
   471	          lastErrStr.includes('API_KEY_INVALID') ||
   472	          lastErrStr.includes('PERMISSION_DENIED') ||
   473	          lastErrStr.includes('UNAUTHENTICATED');
   474	
   475	        if (isDeadKey) {
   476	          break;
   477	        }
   478	      }
   479	    }
   480	
   481	    const errorMsg = String(lastError?.message || lastError || '');
   482	    if (errorMsg.includes('429') || errorMsg.includes('RESOURCE_EXHAUSTED') || errorMsg.includes('Quota exceeded')) {
   483	      const err = new Error('Batas kuota harian/menit AI Gemini telah terlampaui (429 Rate Limit). Silakan tambahkan satu atau beberapa API Key cadangan di menu "Anti Limit API" di bagian atas.');
   484	      (err as any).statusCode = 429;
   485	      throw err;
   486	    } else if (errorMsg.includes('503') || errorMsg.includes('UNAVAILABLE') || errorMsg.includes('demand')) {
   487	      const err = new Error('Model AI Gemini saat ini sedang mengalami lonjakan trafik (503 High Demand). Silakan coba klik tombol "Hasilkan Ulang" dalam beberapa saat lagi.');
   488	      (err as any).statusCode = 503;
   489	      throw err;
   490	    }
   491	
   492	    throw lastError || new Error('Gagal menghasilkan prompt dari AI Gemini.');
   493	  }

  1890	  interface QueueItem {
  1891	    id: number;
  1892	    payload: any;
  1893	    timestamp: number;
  1894	  }
  1895	  const simpleEventQueue: QueueItem[] = [];
  1896	  let globalEventCounter = Date.now();
  1897	
  1898	  function pushToEventQueue(payload: any): number {
  1899	    const eventId = ++globalEventCounter;
  1900	    simpleEventQueue.push({ id: eventId, payload, timestamp: Date.now() });
  1901	    if (simpleEventQueue.length > 500) {
  1902	      simpleEventQueue.shift(); // Evict oldest
  1903	    }
  1904	    return eventId;
  1905	  }

  1907	  // Pending Long-Poll Requests Queue
  1908	  interface PendingPoll {
  1909	    token: string;
  1910	    lastEventId: number;
  1911	    res: any;
  1912	    timeout: any;
  1913	  }

  1915

  1917	  setInterval(() => {
  1918	    const hbPayload = `: heartbeat\nid: hb_${Date.now()}\ndata: ${JSON.stringify({ type: 'ping', ts: Date.now() })}\n\n`;
  1919	    sseClients.forEach((client) => {
  1920	      try {
  1921	        client.res.write(hbPayload);

  1924	      }
  1925	    });

  1928	  // Robust Event Broadcast Function
  1929	  function broadcastLiveEvent(data: any) {
  1930	    const eventId = pushToEventQueue(data);
  1931	    const payload = `:pad\nid: ${eventId}\ndata: ${JSON.stringify(data)}\n\n`;
  1932	
  1933	    // 1. Broadcast to SSE clients
  1934	    sseClients.forEach((client) => {
  1935	      try {
  1936	        client.res.write(payload);
  1937	      } catch (e) {
  1938	        // Retry 1x before dropping
  1939	        setTimeout(() => {
  1940	          try {
  1941	            client.res.write(payload);
  1942	          } catch (retryErr) {
  1943	            logger.warn('[SSE] Client write failed after retry, dropping client.');
  1944	            sseClients.delete(client);
  1945	          }
  1946	        }, 50);
  1947	      }
  1948	    });
  1949	
  1950	    // 2. Resolve Long-Polling clients waiting for new events
  1951	    for (let i = pendingPolls.length - 1; i >= 0; i--) {
  1952	      const poll = pendingPolls[i];
  1953	      if (poll.lastEventId < eventId) {
  1954	        clearTimeout(poll.timeout);
  1955	        try {
  1956	          poll.res.json({
  1957	            success: true,
  1958	            events: [data],
  1959	            lastEventId: eventId,
  1960	          });
  1961	        } catch (e) {
  1962	          // ignore write errors
  1963	        }
  1964	        pendingPolls.splice(i, 1);
  1965	      }
  1966	    }
  1967	  }
